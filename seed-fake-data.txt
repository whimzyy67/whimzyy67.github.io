/* Whimzyy fake data. Paste this whole thing into the browser console (F12 > Console)
   on ANY Whimzyy page, press Enter, and the page reloads with 30 days of data
   and EVERYTHING unlocked (all themes, backgrounds, titles) plus 5000 spare stars.

   It first saves a backup of your real data in "whimzyy-backup", and to undo it
   run the second snippet at the bottom of this file. */
(function () {
  const DAY = 86400000;
  const now = Date.now();
  const key = (ms) => new Date(ms).toLocaleDateString("en-CA");
  const rand = (a, b) => Math.floor(Math.random() * (b - a + 1)) + a;
  const pick = (list) => list[rand(0, list.length - 1)];

  // ---- back up everything that starts with "whimzyy-" (once) ----
  if (!localStorage.getItem("whimzyy-backup")) {
    const backup = {};
    Object.keys(localStorage).forEach((k) => {
      if (k.indexOf("whimzyy-") === 0) { backup[k] = localStorage.getItem(k); }
    });
    localStorage.setItem("whimzyy-backup", JSON.stringify(backup));
  }
  Object.keys(localStorage).forEach((k) => {
    if (k.indexOf("whimzyy-") === 0 && k !== "whimzyy-backup") { localStorage.removeItem(k); }
  });

  const taskNames = ["Read chapter 4", "Finish CSS layout", "Practice JavaScript loops", "Revise logic notes",
    "Fix the Vibe page", "Write journal", "Solve 5 puzzles", "Plan next week", "Watch the grid tutorial",
    "Clean up home.css", "Build the rewards page", "Stretch + water break"];
  const moods = ["slay", "butterflies", "lowkey", "mid", "complicated", "dead"];
  const answers = ["Slow start but I locked in after lunch.", "Good one. Finished everything I planned.",
    "Mid ngl, but I showed up.", "Brain was foggy, did the small stuff.", "Main character energy today!",
    "Tired, but proud I opened the laptop.", "Got stuck on a bug, fixed it at night."];
  const journalTexts = ["Dear Diary,\nToday the layout finally clicked. Grid is not scary anymore.",
    "Dear Diary,\nI keep forgetting to drink water. The reminders help a lot.",
    "Dear Diary,\nStudied late again. Tomorrow I sleep earlier. Allegedly.",
    "Dear Diary,\nSmall win: my to-do list is all ticked."];
  const parkingTexts = ["Look up flex vs grid", "Ask about localStorage limits", "Reply to the email",
    "Buy a new notebook", "Idea: sticker packs as rewards"];

  const sessions = [];
  const tasks = [];
  const journal = [];
  const parking = [];
  const moodMap = {};
  const rewardLog = [];
  const minutesMap = {};
  const goalDays = {};
  const detoxHistory = {};
  const detoxMilestones = {};
  let earned = 0;

  const star = (ts, amount, reason, kind) => {
    rewardLog.push({ ts: ts, day: key(ts), amount: amount, reason: reason, kind: kind });
    earned += amount;
  };

  for (let i = 29; i >= 0; i--) {
    const dayStart = new Date(now - i * DAY);
    dayStart.setHours(0, 0, 0, 0);
    const k = key(dayStart.getTime());
    const isToday = (i === 0);

    // a few quiet days (never the last 9, so the streak is nice)
    if (i > 9 && Math.random() < 0.25) { continue; }

    // ---- sessions ----
    const count = rand(1, 4);
    let dayMinutes = 0;
    let dayTimeStars = 0;
    for (let s = 0; s < count; s++) {
      const hour = pick([7, 9, 10, 11, 14, 15, 16, 19, 20, 22, 23]);
      let start = dayStart.getTime() + hour * 3600000 + rand(0, 50) * 60000;
      if (isToday && start > now - 3600000) { start = now - (s + 2) * 3600000; }
      const mode = pick(["pomodoro", "pomodoro", "timer", "stopwatch"]);
      const mins = mode === "pomodoro" ? pick([25, 25, 50]) : pick([15, 25, 30, 45, 60]);
      const completed = Math.random() > 0.2;
      const real = completed ? mins : Math.max(5, mins - rand(5, 15));
      const session = {
        id: start, start: start, seconds: real * 60, mode: mode,
        task: (mode === "pomodoro" && Math.random() > 0.3) ? pick(taskNames) : "", completed: completed
      };
      if (Math.random() < 0.3) { session.source = "workspace"; }      // some came from Work Space
      sessions.push(session);

      dayMinutes += real;
      let stars = Math.floor(real / 5) + (completed && real >= 15 ? 2 : 0);
      stars = Math.min(stars, 80 - dayTimeStars);
      dayTimeStars += stars;
      if (stars > 0) {
        star(start + real * 60000, stars,
          "Focus · " + { pomodoro: "Pomodoro", timer: "Timer", stopwatch: "Stopwatch" }[mode] + " · " + real + " min" + (completed ? " ✓" : ""), "time");
      }
    }
    minutesMap[k] = dayMinutes;
    if (dayMinutes >= 120) {
      goalDays[k] = 1;
      star(dayStart.getTime() + 20 * 3600000, 5, "Daily goal reached 🎯", "bonus");
    }

    // ---- tasks done ----
    const done = rand(1, 5);
    for (let t = 0; t < done; t++) {
      const doneAt = dayStart.getTime() + rand(9, 21) * 3600000 + rand(0, 59) * 60000;
      if (isToday && doneAt > now) { continue; }
      const text = pick(taskNames);
      tasks.push({ text: text, done: true, doneAt: doneAt, rewarded: true });
      if (t < 10) { star(doneAt, 1, "Task · " + text, "task"); }
    }

    // ---- Detox: screen time for finished days (today stays empty so you can log live) ----
    if (!isToday) {
      const good = Math.random() < (i <= 9 ? 0.85 : 0.6);                     // mostly under the 120 min limit
      const total = good ? rand(45, 115) : rand(125, 210);
      const ig = Math.round(total * 0.35), yt = Math.round(total * 0.4);
      detoxHistory[k] = { total: total, limit: 120, on: true,
        apps: { instagram: ig, youtube: yt, reddit: Math.max(0, total - ig - yt) } };
      if (good) {
        detoxMilestones["detox:" + k] = 1;
        star(dayStart.getTime() + 23 * 3600000, 5, "Detox day under your limit 🌿", "bonus");
      }
    }

    // ---- moods, energy, water, answers ----
    moodMap[k] = pick(moods);
    localStorage.setItem("whimzyy-energy-" + k, String(rand(1, 5)));
    localStorage.setItem("whimzyy-water-" + k, String(rand(2, 8)));
    localStorage.setItem("whimzyy-answer-" + k, pick(answers));

    // ---- journal and parking lot ----
    if (Math.random() > 0.4) {
      const created = dayStart.getTime() + rand(20, 23) * 3600000;
      if (!(isToday && created > now)) {
        journal.push({ id: created, text: pick(journalTexts), created: created, edited: Math.random() > 0.8 ? created + 600000 : 0 });
      }
    }
    if (Math.random() > 0.6) {
      const created = sessions[sessions.length - 1].start + 600000;
      if (created < now) { parking.push({ id: created, text: pick(parkingTexts), created: created, task: "" }); }
    }
  }

  // a few tasks that are not done yet
  ["Finish the login page", "Pick a new theme", "Revise chapter 5"].forEach((text) => tasks.push({ text: text, done: false }));

  // streak bonuses for the last 9 days (a 7-day streak)
  const sevenAgo = new Date(now - 6 * DAY);
  star(new Date(now - DAY).setHours(21, 0, 0, 0), 3, "3-day streak 🔥", "bonus");
  star(now - 3600000, 10, "7-day streak 🔥", "bonus");

  // welcome gift, then a few things bought
  rewardLog.unshift({ ts: sessions.length ? sessions[0].start - 1000 : now, day: key(sessions.length ? sessions[0].start : now), amount: 10, reason: "Welcome gift 🎁", kind: "bonus" });
  earned += 10;

  const spent = [
    ["Unlocked theme · Midnight Plum", 40, "theme:plum"],
    ["Unlocked title · Main Character", 20, "title:Main Character"],
    ["Unlocked background · Sci-Fi Capsule", 25, "bg:3"],
    ["Treat · boba", 30, null]
  ];
  const owned = {};
  let spentTotal = 0;
  spent.forEach((row, i) => {
    const ts = now - (spent.length - i) * 2 * DAY;
    rewardLog.push({ ts: ts, day: key(ts), amount: -row[1], reason: row[0], kind: "spend" });
    spentTotal += row[1];
    if (row[2]) { owned[row[2]] = 1; }
  });
  rewardLog.sort((a, b) => a.ts - b.ts);

  localStorage.setItem("whimzyy-rewards", JSON.stringify({
    v: 1, balance: earned - spentTotal, lifetime: earned, log: rewardLog, owned: owned,
    title: "Main Character",
    treats: [{ id: 1, name: "boba", cost: 30 }, { id: 2, name: "one episode", cost: 20 }, { id: 3, name: "30 min of gaming", cost: 45 }],
    minutes: minutesMap, goalDays: goalDays, milestones: detoxMilestones, welcomed: true
  }));

  // ---- everything else ----
  sessions.sort((a, b) => b.start - a.start);
  journal.sort((a, b) => b.created - a.created);
  parking.sort((a, b) => b.created - a.created);

  localStorage.setItem("whimzyy-sessions", JSON.stringify(sessions));
  localStorage.setItem("whimzyy-tasks", JSON.stringify(tasks));
  localStorage.setItem("whimzyy-journal", JSON.stringify(journal));
  localStorage.setItem("whimzyy-parking-notes", JSON.stringify(parking));
  localStorage.setItem("whimzyy-moods", JSON.stringify(moodMap));
  localStorage.setItem("whimzyy-detox-history", JSON.stringify(detoxHistory));
  localStorage.setItem("whimzyy-goal", "120");
  localStorage.setItem("whimzyy-theme", "plum");
  localStorage.setItem("whimzyy-widgets", JSON.stringify({ clock: true, todo: true, timer: true, message: true }));
  localStorage.setItem("whimzyy-message", "Romanticize the grind, then actually do it.");
  localStorage.setItem("whimzyy-reminders", JSON.stringify({
    water: { on: true, every: 45 }, stretch: { on: false, every: 45 }, rest: { on: true, every: 30 }
  }));

  // ---- unlock everything: every theme (incl. Main Character Drip), background and title + 5000 spare stars ----
  const rw = JSON.parse(localStorage.getItem("whimzyy-rewards"));
  ["wine","plum","moss","navy","rose","sp_mint","sp_grape","sp_matcha","sp_latte","sp_sunset","sp_oldmoney",
   "sp_stealth","sp_honey","sp_pookie","sp_goblin","sp_cobalt","sp_villain","sp_sigma","sp_mauve"]
    .forEach((id) => { rw.owned["theme:" + id] = 1; });
  for (let i = 0; i <= 9; i++) { rw.owned["bg:" + i] = 1; }
  ["Main Character","Absolute Workaholic","Focus Freak","Grind Lord","Efforlessly Elite",
   "Unhinged Icon","Ruthless Beast","Ultimate Rizzler","The GOAT-ed"]
    .forEach((id) => { rw.owned["title:" + id] = 1; });
  rw.balance += 5000;
  rw.lifetime += 5000;
  localStorage.setItem("whimzyy-rewards", JSON.stringify(rw));

  console.log("Fake data ready:", sessions.length, "sessions,", tasks.length, "tasks,", journal.length,
    "journal entries,", Object.keys(detoxHistory).length, "detox days,", rewardLog.length, "star entries,", (earned - spentTotal), "stars to spend.");
  location.reload();
})();
